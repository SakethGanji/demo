-- monthly revenue by plan — in-month rank + running total (CTE + window fns)
WITH monthly AS (
  SELECT date_trunc('month', signup_date) AS month, plan,
         sum(amount) AS revenue, count(*) AS signups
  FROM signups WHERE utm_campaign IS NOT NULL   -- a masked column is never filterable
  GROUP BY 1, 2)
SELECT m.month, m.plan, p.list_price, m.revenue, m.signups,
       rank() OVER (PARTITION BY m.month ORDER BY m.revenue DESC) AS plan_rank,
       sum(m.revenue) OVER (PARTITION BY m.plan ORDER BY m.month
                            ROWS UNBOUNDED PRECEDING) AS running_revenue
FROM monthly m JOIN plan_catalog p ON p.plan_code = m.plan  -- join across sheet_keys
ORDER BY m.month, plan_rank;
