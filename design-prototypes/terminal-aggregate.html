SELECT floor(amount / 500) * 500 AS amount_bin, count(signup_id) AS signups, sum(amount) AS gross_amount,
       sum(amount) FILTER (WHERE status = 'active') AS active_amount, avg(amount) AS avg_ticket, median(amount) AS median_ticket,
       stddev_samp(amount) AS std_ticket, count(DISTINCT utm_campaign) AS campaigns  FROM signups
WHERE signup_date >= DATE '2026-07-01' AND utm_campaign IS NOT NULL AND amount >= 0  -- email is never filterable
GROUP BY 1 HAVING signups > 500 AND gross_amount >= 3000000 ORDER BY gross_amount DESC LIMIT 12;
